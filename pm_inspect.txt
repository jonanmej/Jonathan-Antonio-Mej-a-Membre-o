  const [isShiftActive, setIsShiftActive] = useState(() => localStorage.getItem('shiftActive') === 'true');

  const [isLunchActive, setIsLunchActive] = useState(() => localStorage.getItem('lunchActive') === 'true');

  const [shiftComment, setShiftComment] = useState('');
  const [isFetchingShiftLocation, setIsFetchingShiftLocation] = useState(false);
  const [shiftLocationError, setShiftLocationError] = useState('');

  const handleShiftAction = (action: 'start_shift' | 'end_shift' | 'start_lunch' | 'end_lunch') => {
    setIsFetchingShiftLocation(true);
    setShiftLocationError('');
    
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        setIsFetchingShiftLocation(false);
        const lat = pos.coords.latitude;
        const lng = pos.coords.longitude;
        
        const currentRecords = JSON.parse(localStorage.getItem('timesheet_records') || '[]');
        const activeUser = currentUser || 'Operador Local';
        const activeRecordIndex = currentRecords.findLastIndex((r: any) => r.endTime === null && r.user === activeUser);
        const now = new Date().toISOString();

        if (action === 'start_shift') {
           setIsShiftActive(true);
           setShiftStartTime(now);
           localStorage.setItem('shiftActive', 'true');
           localStorage.setItem('shiftStartTime', now);
           
           currentRecords.push({
             id: Date.now(),
             user: activeUser,
             role: 'Técnico Activo',
             startTime: now,
             endTime: null,
             status: 'En Turno',
             startLocation: { lat, lng },
             comment: shiftComment || undefined
           });
        } else if (action === 'end_shift') {
           setIsShiftActive(false);
           localStorage.setItem('shiftActive', 'false');
           localStorage.removeItem('shiftStartTime');
           setIsLunchActive(false);
           localStorage.setItem('lunchActive', 'false');
           localStorage.removeItem('lunchStartTime');
           
           if (activeRecordIndex !== -1) {
             currentRecords[activeRecordIndex].endTime = now;
             currentRecords[activeRecordIndex].status = 'Finalizado';
             currentRecords[activeRecordIndex].endLocation = { lat, lng };
             if (shiftComment) {
                currentRecords[activeRecordIndex].endComment = shiftComment;
             }
           }
        } else if (action === 'start_lunch') {
           setIsLunchActive(true);
           setLunchStartTime(now);
           localStorage.setItem('lunchActive', 'true');
           localStorage.setItem('lunchStartTime', now);
           
           if (activeRecordIndex !== -1) {
             currentRecords[activeRecordIndex].status = 'En Almuerzo';
           }
        } else if (action === 'end_lunch') {
           setIsLunchActive(false);
           localStorage.setItem('lunchActive', 'false');
           localStorage.removeItem('lunchStartTime');
           
           if (activeRecordIndex !== -1) {
             currentRecords[activeRecordIndex].status = 'En Turno';
           }
        }

        localStorage.setItem('timesheet_records', JSON.stringify(currentRecords));
        setShiftComment(''); // clear comment after action
      },
      (err) => {
        setIsFetchingShiftLocation(false);
        setShiftLocationError('No se pudo obtener la ubicación GPS. Por favor, asegúrese de tener los permisos activados.');
      },
      { enableHighAccuracy: true, timeout: 10000 }
    );
  };
`;
